function openModal(id) { document.getElementById(id).classList.remove('hidden'); }
function closeModal(id) { document.getElementById(id).classList.add('hidden'); }

function clearLocalStorageContext() {
    if(confirm("Completely reset device database adjustments? This wipes local checklist edits.")) {
        localStorage.removeItem(STORAGE_KEY);
        localStorage.removeItem(RECON_REGISTRY_KEY);
        location.reload();
    }
}

function commitToDeviceStorage() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(appState));
    localStorage.setItem(RECON_REGISTRY_KEY, JSON.stringify(managerEdits));
}

function hydrateFromDeviceStorage() {
    const rawState = localStorage.getItem(STORAGE_KEY);
    const rawEdits = localStorage.getItem(RECON_REGISTRY_KEY);
    if(rawEdits) managerEdits = JSON.parse(rawEdits);
    if(rawState) {
        appState = JSON.parse(rawState);
        rebuildTrackingRegistry();
        renderCanvas();
    }
}

function rebuildTrackingRegistry() {
    let registry = {};
    Object.values(appState.vans).forEach(list => {
        list.forEach(p => {
            let clean = p.phone.replace(/\s+/g, '');
            if(clean) registry[clean] = (registry[clean] || 0) + 1;
        });
    });
    appState.trackingRegistry = registry;
}

function processPlaintext() {
    const rawText = document.getElementById('raw-input').value;
    if (!rawText.trim()) return;

    let historicalMemory = {};
    Object.values(appState.vans).forEach(list => {
        list.forEach(p => { historicalMemory[p.phone] = { isPresent: p.isPresent, status: p.status }; });
    });

    const chunks = rawText.split(/={3,}/);
    let currentLeader = null;
    let updatedVans = {};

    chunks.forEach(chunk => {
        const lines = chunk.split('\n').map(l => l.trim()).filter(Boolean);
        for(let line of lines) {
            if(line.toUpperCase().includes('LEADER:')) {
                currentLeader = line.replace(/LEADER:\s*/i, '').trim();
                updatedVans[currentLeader] = [];
                break;
            }
        }
        if(!currentLeader) return;

        lines.forEach(line => {
            const match = line.match(/^\d+\.\s*([^-\[\(]+)-\s*([0-9\s\+]+)/);
            if(match) {
                const originalName = match[1].trim();
                const phone = match[2].replace(/\s+/g, '').trim();
                let isPresent = false;
                let status = 'unconfirmed';
                let displayName = originalName;

                if (managerEdits[phone]) {
                    isPresent = managerEdits[phone].isPresent !== undefined ? managerEdits[phone].isPresent : isPresent;
                    status = managerEdits[phone].status !== undefined ? managerEdits[phone].status : status;
                    displayName = managerEdits[phone].name || originalName;
                }

                updatedVans[currentLeader].push({
                    id: crypto.randomUUID(),
                    name: displayName,
                    phone: phone,
                    isPresent: isPresent,
                    status: status,
                    isReplacement: line.includes('REPLACED:') || line.includes('MARGIN ADDITION')
                });
            }
        });
    });

    appState.vans = updatedVans;
    rebuildTrackingRegistry();
    commitToDeviceStorage();
    closeModal('paste-modal');
    renderCanvas();
}

function triggerCommHub(name, phone) {
    document.getElementById('comm-name').innerText = name;
    document.getElementById('comm-phone').innerText = phone;
    let formatted = phone.startsWith('0') ? '254' + phone.substring(1) : phone.replace('+', '');
    const msgTemplate = encodeURIComponent(`Hey, I'm reaching out regarding your seat assignment for the Grace Encounter dispatch. Let us know if you're en route!`);
    
    document.getElementById('action-wa').href = `https://wa.me{formatted}?text=${msgTemplate}`;
    document.getElementById('action-sms').href = `sms:${phone}?body=${msgTemplate}`;
    document.getElementById('action-tel').href = `tel:${phone}`;
    openModal('comm-modal');
}

function updateInlineField(vanKey, index, field, value) {
    const passenger = appState.vans[vanKey][index];
    passenger[field] = value;
    if(!managerEdits[passenger.phone]) managerEdits[passenger.phone] = {};
    managerEdits[passenger.phone][field] = value;
    commitToDeviceStorage();
    if (field === 'isPresent' || field === 'status') renderCanvas();
}

function swapToReserve(vanKey, index) {
    appState.archives.push(appState.vans[vanKey][index]);
    if(appState.reserves.length > 0) appState.vans[vanKey][index] = appState.reserves.shift();
    else appState.vans[vanKey].splice(index, 1);
    rebuildTrackingRegistry(); commitToDeviceStorage(); renderCanvas();
}

function moveToReserve(vanKey, index) {
    appState.reserves.push(appState.vans[vanKey].splice(index, 1)[0]);
    rebuildTrackingRegistry(); commitToDeviceStorage(); renderCanvas();
}

function promoteFromReserve(vanKey) {
    if(appState.vans[vanKey].length >= MAX_CAPACITY || appState.reserves.length === 0) return;
    appState.vans[vanKey].push(appState.reserves.shift());
    rebuildTrackingRegistry(); commitToDeviceStorage(); renderCanvas();
}

3. app.js (The Logic Component Engine)
javascript
const STORAGE_KEY = "HARAKA_DISPATCH_MATRIX_STATE";
const RECON_REGISTRY_KEY = "HARAKA_MGR_RECONCILIATION";
const MAX_CAPACITY = 14;

let appState = { vans: {}, reserves: [], archives: [], trackingRegistry: {} };
let managerEdits = {};

function openModal(id) { document.getElementById(id).classList.remove('hidden'); }
function closeModal(id) { document.getElementById(id).classList.add('hidden'); }

function clearLocalStorageContext() {
    if(confirm("Completely reset device database adjustments? This wipes local checklist edits.")) {
        localStorage.removeItem(STORAGE_KEY);
        localStorage.removeItem(RECON_REGISTRY_KEY);
        location.reload();
    }
}

function commitToDeviceStorage() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(appState));
    localStorage.setItem(RECON_REGISTRY_KEY, JSON.stringify(managerEdits));
}

function hydrateFromDeviceStorage() {
    const rawState = localStorage.getItem(STORAGE_KEY);
    const rawEdits = localStorage.getItem(RECON_REGISTRY_KEY);
    if(rawEdits) managerEdits = JSON.parse(rawEdits);
    if(rawState) {
        appState = JSON.parse(rawState);
        rebuildTrackingRegistry();
        renderCanvas();
    }
}

function rebuildTrackingRegistry() {
    let registry = {};
    Object.values(appState.vans).forEach(vanData => {
        vanData.passengers.forEach(p => {
            let clean = p.phone.replace(/\s+/g, '');
            if(clean) registry[clean] = (registry[clean] || 0) + 1;
        });
    });
    appState.trackingRegistry = registry;
}

function processPlaintext() {
    const rawText = document.getElementById('raw-input').value;
    if (!rawText.trim()) return;

    let historicalMemory = {};
    Object.values(appState.vans).forEach(vanData => {
        vanData.passengers.forEach(p => { historicalMemory[p.phone] = { isPresent: p.isPresent, status: p.status }; });
    });

    const chunks = rawText.split(/={3,}/);
    let currentLeader = null;
    let updatedVans = {};

    chunks.forEach(chunk => {
        const lines = chunk.split('\n').map(l => l.trim()).filter(Boolean);
        for(let line of lines) {
            if(line.toUpperCase().includes('LEADER:')) {
                currentLeader = line.replace(/LEADER:\s*/i, '').trim();
                // Retain historical parameters for specific vehicle containers if present
                updatedVans[currentLeader] = {
                    pickupTime: appState.vans[currentLeader]?.pickupTime || "06:00 AM",
                    routingLocation: appState.vans[currentLeader]?.routingLocation || "Main Gate Hub",
                    passengers: []
                };
                break;
            }
        }
        if(!currentLeader) return;

        lines.forEach(line => {
            const match = line.match(/^\d+\.\s*([^-\[\(]+)-\s*([0-9\s\+]+)/);
            if(match) {
                const originalName = match[1].trim();
                const phone = match[2].replace(/\s+/g, '').trim();
                let isPresent = false;
                let status = 'unconfirmed';
                let displayName = originalName;

                if (managerEdits[phone]) {
                    isPresent = managerEdits[phone].isPresent !== undefined ? managerEdits[phone].isPresent : isPresent;
                    status = managerEdits[phone].status !== undefined ? managerEdits[phone].status : status;
                    displayName = managerEdits[phone].name || originalName;
                }

                updatedVans[currentLeader].passengers.push({
                    id: crypto.randomUUID(),
                    name: displayName,
                    phone: phone,
                    isPresent: isPresent,
                    status: status,
                    isReplacement: line.includes('REPLACED:') || line.includes('MARGIN ADDITION')
                });
            }
        });
    });

    appState.vans = updatedVans;
    rebuildTrackingRegistry();
    commitToDeviceStorage();
    closeModal('paste-modal');
    renderCanvas();
}

function triggerCommHub(name, phone, vanKey) {
    document.getElementById('comm-name').innerText = name;
    document.getElementById('comm-phone').innerText = phone;
    
    let formatted = phone.startsWith('0') ? '254' + phone.substring(1) : phone.replace('+', '');
    const vanContext = appState.vans[vanKey];
    const time = vanContext?.pickupTime || "06:00 AM";
    const loc = vanContext?.routingLocation || "Main Gate Hub";
    
    const msgTemplate = encodeURIComponent(`Hey, I'm reaching out regarding your seat assignment for the Grace Encounter dispatch. Your matatu lead is ${vanKey}. Setup is set for ${time} at ${loc}. Let us know if you're en route!`);
    
    document.getElementById('action-wa').href = `https://wa.me{formatted}?text=${msgTemplate}`;
    document.getElementById('action-sms').href = `sms:${phone}?body=${msgTemplate}`;
    document.getElementById('action-tel').href = `tel:${phone}`;
    openModal('comm-modal');
}

function updateInlineField(vanKey, index, field, value) {
    const passenger = appState.vans[vanKey].passengers[index];
    passenger[field] = value;
    if(!managerEdits[passenger.phone]) managerEdits[passenger.phone] = {};
    managerEdits[passenger.phone][field] = value;
    commitToDeviceStorage();
    if (field === 'isPresent' || field === 'status') renderCanvas();
}

function updateVanMeta(vanKey, field, value) {
    appState.vans[vanKey][field] = value;
    commitToDeviceStorage();
}

function swapToReserve(vanKey, index) {
    appState.archives.push(appState.vans[vanKey].passengers[index]);
    if(appState.reserves.length > 0) appState.vans[vanKey].passengers[index] = appState.reserves.shift();
    else appState.vans[vanKey].passengers.splice(index, 1);
    rebuildTrackingRegistry(); commitToDeviceStorage(); renderCanvas();
}

function moveToReserve(vanKey, index) {
    appState.reserves.push(appState.vans[vanKey].passengers.splice(index, 1)[0]);
    rebuildTrackingRegistry(); commitToDeviceStorage(); renderCanvas();
}

function promoteFromReserve(vanKey) {
    if(appState.vans[vanKey].passengers.length >= MAX_CAPACITY || appState.reserves.length === 0) return;
    appState.vans[vanKey].passengers.push(appState.reserves.shift());
    rebuildTrackingRegistry(); commitToDeviceStorage(); renderCanvas();
}

    const keys = Object.keys(appState.vans);

    }
    document.getElementById('empty-state').classList.add('hidden');

    keys.forEach(leader => {
        const vanContext = appState.vans[leader];
        const passengers = vanContext.passengers;
        totalAssigned += passengers.length;
        
        // Calculate Load Percentage metrics dynamically
        const loadPercentage = Math.min(Math.round((passengers.length / MAX_CAPACITY) * 100), 100);
        const barColor = loadPercentage >= 100 ? 'bg-rose-500' : loadPercentage >= 75 ? 'bg-amber-500' : 'bg-emerald-500';

        const card = document.createElement('div');
        card.className = "bg-zinc-900/90 border border-zinc-800/80 rounded-xl p-3 flex flex-col max-h-[520px]";
        
        let passengerRows = passengers.map((p, idx) => {
            let cleanPhone = p.phone.replace(/\s+/g, '');
            let isDuplicate = (appState.trackingRegistry[cleanPhone] || 1) > 1;
            let bgStyle = p.status === 'confirmed' ? 'bg-emerald-950/10 border-emerald-800/30' : p.status === 'declined' ? 'bg-rose-950/10 border-rose-800/30' : 'bg-zinc-950/60 border-zinc-800/50';

            return `

                    <div class="flex items-center justify-between gap-1">

                            <input type="text" value="${p.name}" oninput="updateInlineField('${leader}', ${idx}, 'name', this.value)" class="bg-transparent border-none p-0 focus:ring-0 font-medium text-xs text-zinc-200 truncate w-full focus:outline-none">

                            <option value="declined" ${p.status === 'declined' ? 'selected' : ''}>❌ Declined</option>
                        </select>
                    </div>
                </div>`;
        }).join('');
